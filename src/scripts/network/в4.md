class_name PacketTypes

# ── Типы сообщений ────────────────────────────────
enum {
	CONNECT       = 1,
	DISCONNECT    = 2,
	PING          = 3,
	PONG          = 4,
	PLAYER_SPAWN  = 5,   # server → new client: «ты — id N, позиция X»
	PLAYER_JOINED = 6,   # server → clients:    «подключился игрок N»
	PLAYER_LEFT   = 7,   # server → clients:    «игрок N ушёл»
	PLAYER_UPDATE = 8,   # position + rotation  (unreliable, 20 Hz)
}

# ── Размер заголовка ──────────────────────────────
const HEADER_SIZE := 3   # 1 type + 2 body_length


# ══════════════════════════════════════════════════
#  ОБЩИЕ: пакет = заголовок + тело
# ══════════════════════════════════════════════════

static func write_packet(type: int, body: PackedByteArray = PackedByteArray()) -> PackedByteArray:
	var buf := StreamPeerBuffer.new()
	buf.big_endian = false
	buf.put_u8(type)
	buf.put_u16(body.size())
	if body.size() > 0:
		buf.put_data(body)
	return buf.data_array


static func read_packet(raw: PackedByteArray) -> Dictionary:
	if raw.size() < HEADER_SIZE:
		push_warning("PacketTypes: пакет слишком короткий (%d байт)" % raw.size())
		return { "type": -1, "body": null }

	var r := StreamPeerBuffer.new()
	r.big_endian = false
	r.data_array = raw
	r.seek(0)

	var pkt_type  : int = r.get_u8()
	var body_len  : int = r.get_u16()

	if raw.size() < HEADER_SIZE + body_len:
		push_warning("PacketTypes: ожидалось %d байт, получено %d"
				% [HEADER_SIZE + body_len, raw.size()])
		return { "type": -1, "body": null }

	var body_buf := StreamPeerBuffer.new()
	body_buf.big_endian = false
	if body_len > 0:
		body_buf.data_array = raw.slice(HEADER_SIZE, HEADER_SIZE + body_len)
	body_buf.seek(0)
	return { "type": pkt_type, "body": body_buf }


# ══════════════════════════════════════════════════
#  ПРИВАТНЫЙ ХЕЛПЕР: тело «состояние игрока»
#    peer_id(u32) + pos(3×f32) + rot(3×f32) = 28 байт
# ══════════════════════════════════════════════════

static func _write_player_body(peer_id: int, pos: Vector3, rot: Vector3) -> PackedByteArray:
	var b := StreamPeerBuffer.new()
	b.big_endian = false
	b.put_u32(peer_id)
	b.put_float(pos.x); b.put_float(pos.y); b.put_float(pos.z)
	b.put_float(rot.x); b.put_float(rot.y); b.put_float(rot.z)
	return b.data_array


static func _read_player_body(buf: StreamPeerBuffer) -> Dictionary:
	var id  := buf.get_u32()
	var pos := Vector3(buf.get_float(), buf.get_float(), buf.get_float())
	var rot := Vector3(buf.get_float(), buf.get_float(), buf.get_float())
	return { "peer_id": id, "position": pos, "rotation": rot }


# ══════════════════════════════════════════════════
#  PLAYER_SPAWN
# ══════════════════════════════════════════════════

static func write_player_spawn(peer_id: int, pos: Vector3, rot: Vector3) -> PackedByteArray:
	return write_packet(PLAYER_SPAWN, _write_player_body(peer_id, pos, rot))

static func read_player_spawn(buf: StreamPeerBuffer) -> Dictionary:
	return _read_player_body(buf)


# ══════════════════════════════════════════════════

static func write_player_joined(peer_id: int, pos: Vector3, rot: Vector3) -> PackedByteArray:
	return write_packet(PLAYER_JOINED, _write_player_body(peer_id, pos, rot))

static func read_player_joined(buf: StreamPeerBuffer) -> Dictionary:

# ══════════════════════════════════════════════════
#  PLAYER_LEFT
# ══════════════════════════════════════════════════

static func write_player_left(peer_id: int) -> PackedByteArray:
	var b := StreamPeerBuffer.new()
	b.big_endian = false
	b.put_u32(peer_id)
	return write_packet(PLAYER_LEFT, b.data_array)

static func read_player_left(buf: StreamPeerBuffer) -> Dictionary:
	return { "peer_id": buf.get_u32() }


# ══════════════════════════════════════════════════

static func write_player_update(peer_id: int, pos: Vector3, rot: Vector3) -> PackedByteArray:
